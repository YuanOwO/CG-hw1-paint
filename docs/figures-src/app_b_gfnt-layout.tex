        \def\x{0}%
        \foreach \name/\bytes/\col in {#2} {%
            \pgfmathsetmacro{\w}{\bytes*\unit}%
            \node[field, fill=\col, minimum width=\w cm, anchor=north west]
                at (\x, #1) {\name\\[-1pt]{\scriptsize\color{black!55}\bytes\ B}};%
            \pgfmathsetmacro{\x}{\x+\w}%
            \global\let\x\x%
        }%
    }
